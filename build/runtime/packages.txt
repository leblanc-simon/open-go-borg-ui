# Paquets Cygwin du runtime de sauvegarde, un par ligne, commentaires ignorés.
#
# Deux listes distinctes, parce que les outils de compilation pèsent plus lourd
# que tout le reste : la première sert à construire Borg, la seconde compose le
# runtime réellement livré. Une entrée ajoutée à « release » sans nécessité
# démontrée est une entrée à retirer.

[build]
# Chaîne de compilation et en-têtes : Borg n'a pas de roue précompilée pour
# Cygwin, il est compilé depuis les sources.
#
# La série de Python est celle que Cygwin distribue (méta-paquet python3 =
# python312 à ce jour) et doit rester dans les versions acceptées par Borg :
# 1.4.5 exige Python 3.10 ou plus récent. La liste python39 que donne la
# documentation d'installation de Borg n'est plus valide.
gcc-g++
binutils
make
python312
python312-devel
python312-pip
python312-setuptools
python312-wheel
libssl-devel
liblz4-devel
libzstd-devel
libxxhash-devel

[release]
# Les ACL POSIX ne demandent aucun paquet : Cygwin les fournit dans sa propre
# bibliothèque C. Il n'existe pas de libacl à installer, contrairement à une
# distribution Linux.
#
# Le strict nécessaire à l'exécution : l'interpréteur, les bibliothèques de
# compression et de chiffrement, le client SSH et le shell par lequel
# l'application lance ses commandes.
python312
openssh
bash
coreutils
libssl3
liblz4_1
libzstd1
libxxhash0
